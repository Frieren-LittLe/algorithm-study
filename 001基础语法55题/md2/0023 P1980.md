## 完整代码

```cpp
#include <iostream>
using namespace std;

int main() {
    int n, x, ans = 0;
    cin >> n >> x;

    for (int i = 1; i <= n; i++) {

        int t = i;

        while (t > 0) {
            if (t % 10 == x) {

                ans++;

            }
            t = t / 10;

        }
    }

    cout << ans << endl;
    return 0;
}
```
