## 完整代码

```cpp
#include<iostream>
using namespace std;
int main() {
	int n;
	int a[101];
	cin >> n;
	for (int i = 0;i < n;i++) {
		cin >> a[i];

	}
	int min_a = a[0];
	int max_a = a[0];

	for (int i = 1;i < n;i++) {
		if (a[i] < min_a) {
			min_a = a[i];

		}
		if (a[i] > max_a) {
			max_a = a[i];

		}
	}
	cout << max_a - min_a << endl;
	return 0;
}
```
