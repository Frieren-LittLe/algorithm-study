## 完整代码

```cpp
#include<iostream>
using namespace std;

int main() {
	int a[100];

	int n = 0;

	int x;
	while (cin >> x) {

		if (x == 0)break;

		a[n] = x;

		n++;
	}
	for (int i = n - 1;i >= 0;i--) {

		cout << a[i] << " ";
	}
	return 0;
}
```
