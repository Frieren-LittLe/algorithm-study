## 完整代码

```cpp
#include<iostream>
using namespace std;

int main() {
	int n;
	cin >> n;
	int a[100];
	for (int i = 0;i < n;i++) {
		cin >> a[i];
	}

	for (int i = 0;i < n;i++) {
		int L = 0;

		for (int j = 0;j < i;j++)

			if (a[j] < a[i]) {
				L++;

			}
		cout << L << " ";
	}
	return 0;
}
```
