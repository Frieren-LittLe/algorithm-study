## 完整代码

```cpp
#include<iostream>
#include<climits>

using namespace std;
int main() {
	int n, m;
	cin >> n >> m;
	int a[3000];
	int b[3000];
	int min_b = INT_MAX;

	for (int i = 0; i < n;i++) {
		cin >> a[i];

	}
	for (int i = 0;i <= n - m;i++) {

		b[i] = 0;

		for (int j = 0; j < m;j++) {
			b[i] += a[i + j];

		}
		if (b[i] < min_b) {
			min_b = b[i];

		}
	}
	cout << min_b << endl;
	return 0;
}
```
