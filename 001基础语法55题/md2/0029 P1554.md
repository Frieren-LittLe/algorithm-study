## 完整代码

```cpp
#include<iostream>
using namespace std;

int main() {
	int M, N;
	cin >> M >> N;
	long long cnt[10] = { 0 };

	for (int i = M;i <= N;i++) {
		int t = i;

		while (t > 0) {
			int x = t % 10;
			cnt[x]++;

			t = t / 10;

		}
	}
	for (int i = 0;i < 10;i++) {
		cout << cnt[i] << " ";

	}
}
```
