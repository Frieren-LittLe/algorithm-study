## 完整代码

```cpp
#include <iostream>
using namespace std;

int main() {
	int a[105];
	int b[20000] = { 0 };

	int n
	int ans = 0;
	cin >> n;
	for (int i = 0;i < n;i++) cin >> a[i];
	for (int i = 0;i < n;i++) {

		for (int j = i + 1; j < n; j++) {

			b[a[i] + a[j]] = 1;

		}
	}
	for (int i = 0; i < n; i++) {
		if (b[a[i]] == 1)ans++;

	}
	cout << ans << endl;
	return 0;
}
```
