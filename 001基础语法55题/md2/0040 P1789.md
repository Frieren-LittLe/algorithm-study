## 完整代码

```cpp
#include<iostream>
using namespace std;

int map[102][102];

int main() {
	int n, m, k;
	cin >> n >> m >> k;

	for (int i = 0; i < m;i++) {
		int x, y;
		cin >> x >> y;

		for (int dx = -2;dx <= 2;dx++) {
			for (int dy = -2;dy <= 2;dy++) {
				if (abs(dx) + abs(dy) <= 2) {

					int nx = x + dx;
					int ny = y + dy;

					if (nx >= 1 && nx <= n && ny >= 1 && ny <= n) {

						map[nx][ny] = 1;
					}
				}
			}
		}
	}
	for (int i = 0; i < k; i++) {
		int x, y;
		cin >> x >> y;

		for (int dx = -2; dx <= 2;dx++) {
			for (int dy = -2;dy <= 2;dy++) {
				int nx = x + dx;
				int ny = y + dy;

				if (nx >= 1 && nx <= n && ny >= 1 && ny <= n) {
					map[nx][ny] = 1;
				}
			}
		}
	}

	int ans = 0;
	for (int i = 1; i <= n;i++) {
		for (int j = 1;j <= n;j++) {
			if (map[i][j] == 0) {
				ans++;

			}
		}
	}
	cout << ans << endl;
	return 0;
}
```
