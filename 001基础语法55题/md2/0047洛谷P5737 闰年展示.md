## 完整代码

```cpp
#include<iostream>
using namespace std;

bool Leap(int year) {
	if (year % 400 == 0) return true;
	if (year % 100 == 0) return false;
	if (year % 4 == 0) return true;

	return false;
}

int main() {
	int x, y;
	cin >> x >> y;

	int cnt = 0;
	int a[3000];

	for (int i = x;i <= y; i++) {
		if (Leap(i)) {
			a[cnt] = i;
			cnt++;
		}

	}
	cout << cnt << endl;

	for (int i = 0; i < cnt; i++) {
		cout << a[i] << " ";
	}

	cout << endl;
	return 0;
}
```
