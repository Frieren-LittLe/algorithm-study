## 完整代码

```cpp
#include<iostream>
#include<cmath>
#include<iomanip>
using namespace std;

int Max(int a, int b, int c) {
	int max = a;
	if (b > max) max = b;
	if (c > max) max = c;
	return max;
}

double Domath(int a, int b, int c) {

	int up = Max(a, b, c);

	int down = Max(a + b, b, c) * Max(a, b, b + c);
	double m = (double)up / down;

	return m;
}


int main() {

	int a, b, c;
	cin >> a >> b >> c;
	double m = Domath(a, b, c);
	cout << fixed << setprecision(3) << m << endl;
	return 0;
}
```
