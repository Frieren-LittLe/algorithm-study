## 完整代码

```cpp
#include<iostream>
#include<iomanip>
using namespace std;

double f(double x, int n) {
	if (n == 1) {
		return x / (1 + x);

	}
	return x / (n + f(x, n - 1));

}

int main() {
	double x;
	int n;
	cin >> x >> n;
	cout << fixed << setprecision(2) << f(x, n) << endl;
	return 0;
}
```
