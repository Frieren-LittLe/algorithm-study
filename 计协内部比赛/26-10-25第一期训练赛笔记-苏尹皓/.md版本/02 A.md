# 02 A · 乘 2 除以 6

```cpp
#include <iostream>
using namespace std;
int t;
int n;

int main() {
    cin >> t;
    while (t--) {
        cin >> n;
        int ans = 0;
        if (n == 1) {
            cout << ans << endl;
            continue;
        }
        if (n > 1) {
            int x = n;
            while (x % 6 == 0) {
                x = x / 6;
                ans++;
            }
            while (x % 3 == 0) {
                x = x / 3;
                ans += 2;
            }
            if (x != 1) {
                cout << "-1" << endl;
            } else {
                cout << ans << endl;
            }
        }
    }
    return 0;
}
```
