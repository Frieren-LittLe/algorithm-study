# 05 F · 红蓝对抗

```cpp
#include <iostream>
using namespace std;

int main() {
    int t;
    cin >> t;

    while (t--) {
        int n, r, b;
        cin >> n >> r >> b;

        // b 个 'B' 会把字符串分成 b + 1 段
        int q = r / (b + 1);   // 每一段最少要放几个 'R'
        int rem = r % (b + 1); // 还剩下几个多余的 'R'

        // 循环处理这 b + 1 段
        for (int i = 0; i < b + 1; i++) {
            // 先确定当前这一段要打印几个 'R'
            int count = q;
            if (i < rem) {
                count = count + 1; // 前 rem 段，每段多分 1 个剩下的 'R'
            }

            // 用最基础的循环逐个打印 'R'
            for (int j = 0; j < count; j++) {
                cout << 'R';
            }

            // 在每段后面打印一个 'B'
            // 可以发现由于A的数量多余B，每段至少有一个B，至多只有两个B
            if (i < b) {
                cout << 'B';
            }
        }
        cout << endl;
    }

    return 0;
}
```
