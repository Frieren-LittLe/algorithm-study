# 08 H · 二进制双端队列

```cpp
#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

// 最小操作数 = 总长度 - 留下来的最长连续子数组长度
// 与其计算“最少需要从两端移除多少个元素”
// 不如反过来求“和正好等于s的最长连续子数组”

// 凡是涉及"寻找满足特定条件的连续子数组（或区间）"的问题
// 通常都可以优先考虑使用"滑动窗口（同向双指针）"算法
// 滑动窗口法其实是一种优化过的枚举法

void solve() {
    int n, s;
    cin >> n >> s;

    vector<int> a(n);
    int sum = 0;

    // 特殊情况：
    // 如果全部元素加起来都不够 s，绝对不可能达成
    // 如果总和刚好等于 s，说明一个元素都不用删

    for (int i = 0; i < n; i++) {
        cin >> a[i];
        sum += a[i];
    }

    if (sum < s) {
        cout << -1 << "\n";
        return; // retrun用于跳出分函数
    }

    if (sum == s) {
        cout << 0 << "\n";
        return;
    }

    int L = 0;
    int cur_s = 0;    // 用于记录当前和 （判定的数据）
    int max_len = -1; // 用于记录最大长度(需求的数据）

    // 窗口右侧不断吞入新元素
    // 当窗口内的和超过 s 时，左侧吐出元素，直到和恢复合法
    // 如果和正好等于 s，看看能不能刷新最长记录

    for (int R = 0; R < n; R++) {
        cur_s += a[R];

        while (cur_s > s && L <= R) {
            cur_s -= a[L];
            L++;
        }

        if (cur_s == s) {
            max_len = max(max_len, R - L + 1);
        }
    }
    cout << n - max_len << "\n";
}

int main() {
    // 优化标准输入输出速度
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);

    int t;
    cin >> t;
    while (t--) {
        solve();
    }

    return 0;
}
```
